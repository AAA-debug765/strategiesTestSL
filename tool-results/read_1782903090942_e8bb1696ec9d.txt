    60→function formatElapsed(ms: number): string {
    61→  if (ms < 1000) return `${ms}ms`;
    62→  const s = Math.floor(ms / 1000);
    63→  if (s < 60) return `${s}s`;
    64→  const m = Math.floor(s / 60);
    65→  return `${m}m ${s % 60}s`;
    66→}

    68→const STEP_OPTIONS = [1, 0.1, 0.01, 0.001];

    70→function decimalsFromStep(step: number): number {
    71→  const s = String(step);
    72→  const dot = s.indexOf(".");
    73→  return dot === -1 ? 0 : s.length - dot - 1;
    74→}

    76→// ── WheelInput (non-passive wheel to change value) ────
    77→
    78→function WheelInput({
    79→  value,
    80→  onChange,
    81→  step,
    82→  min,
    83→  max,
    84→  className,
    85→  placeholder,
    86→}: {
    87→  value: number;
    88→  onChange: (v: number) => void;
    89→  step: number;
    90→  min?: number;
    91→  max?: number;
    92→  className?: string;
    93→  placeholder?: string;
    94→}) {
    95→  const inputRef = useRef<HTMLInputElement>(null);
    96→  const decimals = decimalsFromStep(step);
    97→
    98→  const displayVal = value === 0 ? "" : value.toFixed(decimals);
    99→
   100→  const clamp = useCallback((v: number) => {
   101→    let clamped = Math.round(v * Math.pow(10, decimals)) / Math.pow(10, decimals);
   102→    if (min != null) clamped = Math.max(min, clamped);
   103→    if (max != null) clamped = Math.min(max, clamped);
   104→    return clamped;
   105→  }, [decimals, min, max]);
   106→
   107→  const onWheel = useCallback((e: WheelEvent) => {
   108→    e.preventDefault();
   109→    const delta = e.deltaY < 0 ? step : -step;
   110→    const newVal = clamp(value + delta);
   111→    if (newVal !== value) onChange(newVal);
   112→  }, [value, step, clamp, onChange]);
   113→
   114→  useEffect(() => {
   115→    const el = inputRef.current;
   116→    if (!el) return;
   117→    el.addEventListener("wheel", onWheel, { passive: false });
   118→    return () => el.removeEventListener("wheel", onWheel);
   119→  }, [onWheel]);
   120→
   121→  return (
   122→    <input
   123→      ref={inputRef}
   124→      type="text"
   125→      inputMode="decimal"
   126→      value={displayVal}
   127→      placeholder={placeholder}
   128→      onChange={(e) => {
   129→        const raw = e.target.value.replace(/[^0-9.]/g, "");
   130→        if (raw === "" || raw === ".") { onChange(0); return; }
   131→        const parsed = parseFloat(raw);
   132→        if (!isNaN(parsed)) onChange(clamp(parsed));
   133→      }}
   134→      onFocus={(e) => e.target.select()}
   135→      suppressHydrationWarning
   136→      className={`h-5 text-[10px] w-[52px] px-1 bg-background border border-input rounded outline-none focus:ring-1 focus:ring-ring ${className ?? ""}`}
   137→    />
   138→  );
   139→}

   160→function OptimizeProgress() {
   161→  const { optProgress, isOptimizing, funnelMode } = useStrategy();
   162→  const [now, setNow] = useState(Date.now());
   163→
   164→  useEffect(() => {
   165→    if (!isOptimizing) return;
   166→    const timer = setInterval(() => setNow(Date.now()), 500);
   167→    return () => clearInterval(timer);
   168→  }, [isOptimizing]);
   169→
   170→  const { current, total, startTime } = optProgress;
   171→  if (!isOptimizing || total === 0) return null;
   172→
   173→  const remaining = total - current;
   174→  const pct = Math.round((current / total) * 100);
   175→  const elapsed = startTime ? now - startTime : 0;
   176→  const rate = elapsed > 0 ? current / (elapsed / 1000) : 0;
   177→  const eta = rate > 0 ? remaining / rate : 0;
   178→
   179→  return (
   180→    <div className="space-y-1.5 p-2 rounded-md bg-muted/50 border border-border/50">
   181→      <div className="flex items-center justify-between">
   182→        <div className="flex items-center gap-2 text-[10px]">
   183→          <Loader2 className="h-3 w-3 animate-spin text-yellow-500" />
   184→          <span className="text-muted-foreground">
   185→            {current.toLocaleString()} / {total.toLocaleString()}
   186→            <span className="ml-1 text-blue-400">({remaining.toLocaleString()} left)</span>
   187→          </span>
   188→          <span className="font-mono font-medium">{pct}%</span>
   189→          {funnelMode && (
   190→            <span className="text-[9px] px-1.5 py-0.5 rounded bg-blue-500/20 text-blue-300 font-medium">Funnel</span>
   191→          )}
   192→          {optProgress.phase && (
   193→            <span className="text-[9px] text-muted-foreground font-medium">{optProgress.phase}</span>
   194→          )}
   195→        </div>
   196→        <div className="flex items-center gap-2 text-[10px] text-muted-foreground">
   197→          <span className="flex items-center gap-0.5">
   198→            <Clock className="h-2.5 w-2.5" />
   199→            {formatElapsed(elapsed)}
   200→          </span>
   201→          {eta > 1 && (
   202→            <span className="flex items-center gap-0.5">
   203→              <Zap className="h-2.5 w-2.5" />
   204→              ~{formatElapsed(eta * 1000)}
   205→            </span>
   206→          )}
   207→          {rate > 0 && (
   208→            <span className="font-mono">{rate.toFixed(0)}/s</span>
   209→          )}
   210→        </div>
   211→      </div>
   212→      <div className="h-1.5 w-full bg-muted rounded-full overflow-hidden">
   213→        <div
   214→          className="h-full bg-gradient-to-r from-yellow-500 to-orange-500 rounded-full transition-all duration-300 ease-out"
   215→          style={{ width: `${pct}%` }}
   216→        />
   217→      </div>
   218→    </div>
   219→  );
   220→}
