    51→function fmtPct(n: number, d = 2) { return `${n >= 0 ? "+" : ""}${n.toFixed(d)}%`; }
    52→function fmtUsdt(n: number) {
    53→  const abs = Math.abs(n);
    54→  if (abs >= 1000) return `${n >= 0 ? "+" : "-"}$${abs.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
    55→  if (abs >= 1) return `${n >= 0 ? "+" : "-"}$${abs.toFixed(2)}`;
    56→  return `${n >= 0 ? "+" : "-"}$${abs.toFixed(4)}`;
    57→}
    58→function fmtPrice(n: number) {
    59→  if (n >= 1000) return n.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    60→  if (n >= 1) return n.toFixed(4);
    61→  return n.toFixed(6);
    62→}
    63→function formatElapsed(ms: number): string {
    64→  if (ms < 1000) return `${ms}ms`;
    65→  const s = Math.floor(ms / 1000);
    66→  if (s < 60) return `${s}s`;
    67→  const m = Math.floor(s / 60);
    68→  return `${m}m ${s % 60}s`;
    69→}

    73→function decimalsFromStep(step: number): number {
    74→  const s = String(step);
    75→  const dot = s.indexOf(".");
    76→  return dot === -1 ? 0 : s.length - dot - 1;
    77→}

    79→// ── WheelInput (non-passive wheel to change value) ────

    81→function WheelInput({
    82→  value,
    83→  onChange,
    84→  step,
    85→  min,
    86→  max,
    87→  className,
    88→  placeholder,
    89→}: {
    90→  value: number;
    91→  onChange: (v: number) => void;
    92→  step: number;
    93→  min?: number;
    94→  max?: number;
    95→  className?: string;
    96→  placeholder?: string;
    97→}) {
    98→  const inputRef = useRef<HTMLInputElement>(null);
    99→  const decimals = decimalsFromStep(step);
   100→
   101→  const displayVal = value === 0 ? "" : value.toFixed(decimals);
   102→
   103→  const clamp = useCallback((v: number) => {
   104→    let clamped = Math.round(v * Math.pow(10, decimals)) / Math.pow(10, decimals);
   105→    if (min != null) clamped = Math.max(min, clamped);
   106→    if (max != null) clamped = Math.min(max, clamped);
   107→    return clamped;
   108→  }, [decimals, min, max]);
   109→
   110→  const onWheel = useCallback((e: WheelEvent) => {
   111→    e.preventDefault();
   112→    const delta = e.deltaY < 0 ? step : -step;
   113→    const newVal = clamp(value + delta);
   114→    if (newVal !== value) onChange(newVal);
   115→  }, [value, step, clamp, onChange]);
   116→
   117→  useEffect(() => {
   118→    const el = inputRef.current;
   119→    if (!el) return;
   120→    el.addEventListener("wheel", onWheel, { passive: false });
   121→    return () => el.removeEventListener("wheel", onWheel);
   122→  }, [onWheel]);
   123→
   124→  return (
   125→    <input
   126→      ref={inputRef}
   127→      type="text"
   128→      inputMode="decimal"
   129→      value={displayVal}
   130→      placeholder={placeholder}
   131→      onChange={(e) => {
   132→        const raw = e.target.value.replace(/[^0-9.]/g, "");
   133→        if (raw === "" || raw === ".") { onChange(0); return; }
   134→        const parsed = parseFloat(raw);
   135→        if (!isNaN(parsed)) onChange(clamp(parsed));
   136→      }}
   137→      onFocus={(e) => e.target.select()}
   138→      suppressHydrationWarning
   139→      className={`h-5 text-[10px] w-[52px] px-1 bg-background border border-input rounded outline-none focus:ring-1 focus:ring-ring ${className ?? ""}`}
   140→    />
   141→  );
   142→}

   163→function OptimizeProgress() {
   164→  const { optProgress, isOptimizing, funnelMode } = useStrategy();
   165→  const [now, setNow] = useState(Date.now());
   166→
   167→  useEffect(() => {
   168→    if (!isOptimizing) return;
   169→    const timer = setInterval(() => setNow(Date.now()), 500);
   170→    return () => clearInterval(timer);
   171→  }, [isOptimizing]);
   172→
   173→  const { current, total, startTime } = optProgress;
   174→  if (!isOptimizing || total === 0) return null;
   175→
   176→  const remaining = total - current;
   177→  const pct = Math.round((current / total) * 100);
   178→  const elapsed = startTime ? now - startTime : 0;
   179→  const rate = elapsed > 0 ? current / (elapsed / 1000) : 0;
   180→  const eta = rate > 0 ? remaining / rate : 0;
   181→
   182→  return (
   183→    <div className="space-y-1.5 p-2 rounded-md bg-muted/50 border border-border/50">
   184→      <div className="flex items-center justify-between">
   185→        <div className="flex items-center gap-2 text-[10px]">
   186→          <Loader2 className="h-3 w-3 animate-spin text-yellow-500" />
   187→          <span className="text-muted-foreground">
   188→            {current.toLocaleString()} / {total.toLocaleString()}
   189→            <span className="ml-1 text-blue-400">({remaining.toLocaleString()} left)</span>
   190→          </span>
   191→          <span className="font-mono font-medium">{pct}%</span>
   192→          {funnelMode && (
   193→            <span className="text-[9px] px-1.5 py-0.5 rounded bg-blue-500/20 text-blue-300 font-medium">Funnel</span>
   194→          )}
   195→          {optProgress.phase && (
   196→            <span className="text-[9px] text-muted-foreground font-medium">{optProgress.phase}</span>
   197→          )}
   198→        </div>
   199→        <div className="flex items-center gap-2 text-[10px] text-muted-foreground">
   200→          <span className="flex items-center gap-0.5">
   201→            <Clock className="h-2.5 w-2.5" />
   202→            {formatElapsed(elapsed)}
   203→          </span>
   204→          {eta > 1 && (
   205→            <span className="flex items-center gap-0.5">
   206→              <Zap className="h-2.5 w-2.5" />
   207→              ~{formatElapsed(eta * 1000)}
   208→            </span>
   209→          )}
   210→          {rate > 0 && (
   211→            <span className="font-mono">{rate.toFixed(0)}/s</span>
   212→          )}
   213→        </div>
   214→      </div>
   215→      <div className="h-1.5 w-full bg-muted rounded-full overflow-hidden">
   216→        <div
   217→          className="h-full bg-gradient-to-r from-yellow-500 to-orange-500 rounded-full transition-all duration-300 ease-out"
   218→          style={{ width: `${pct}%` }}
   219→        />
   220→      </div>
   221→    </div>
   222→  );
   223→}
